import { SnykService } from './SnykService';
import { OmniNexus } from './OmniNexus';

type ScanInterval = 'daily' | 'weekly' | 'monthly';

class SchedulerServiceClass {
  private intervalId: NodeJS.Timeout | null = null;
  private nextScanTime: Date | null = null;
  private interval: ScanInterval | null = null;

  scheduleSecurityScan(interval: ScanInterval): void {
    this.cancelSchedule();
    this.interval = interval;

    const ms = this.getIntervalMs(interval);
    this.intervalId = setInterval(async () => {
      const result = await SnykService.quickScan();
      if (result && (result.summary.critical > 0 || result.summary.high > 0)) {
        OmniNexus.emit({
          id: `scheduled-scan-${Date.now()}`,
          source: 'security-scheduler',
          priority: 'high',
          message: `?? ??????????????寧?????${result.summary.critical + result.summary.high} ?????????????`,
          timestamp: Date.now(),
          metadata: { scheduled: true, interval, summary: result.summary },
        });
      }
    }, ms);

    this.nextScanTime = new Date(Date.now() + ms);
    localStorage.setItem(
      'scheduledScan',
      JSON.stringify({ interval, nextScan: this.nextScanTime })
    );
  }

  cancelSchedule(): void {
    if (this.intervalId) {
      clearInterval(this.intervalId);
      this.intervalId = null;
      this.nextScanTime = null;
      localStorage.removeItem('scheduledScan');
    }
  }

  getNextScanTime(): Date | null {
    return this.nextScanTime;
  }

  isActive(): boolean {
    return this.intervalId !== null;
  }

  private getIntervalMs(interval: ScanInterval): number {
    switch (interval) {
      case 'daily':
        return 24 * 60 * 60 * 1000;
      case 'weekly':
        return 7 * 24 * 60 * 60 * 1000;
      case 'monthly':
        return 30 * 24 * 60 * 60 * 1000;
    }
  }

  // ??????豯斤??????????重構???????
  restoreSchedule(): void {
    const saved = localStorage.getItem('scheduledScan');
    if (saved) {
      const { interval } = JSON.parse(saved);
      this.scheduleSecurityScan(interval);
    }
  }
}

export const SchedulerService = new SchedulerServiceClass();
